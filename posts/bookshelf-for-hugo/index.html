<script>
(function () {
  /* ─────────────────────────────────────────────
   *  BSML  –  BookShelf Markup Language  parser
   *
   *  Grammar (informal):
   *
   *    bsml        = line* EOF
   *    line        = comment | shelf-decl | book-block | blank
   *    comment     = "#" .* NEWLINE
   *    shelf-decl  = "shelf" WS quoted-str NEWLINE
   *    book-block  = "book" WS quoted-str WS? "{" NEWLINE
   *                  field*
   *                  "}" NEWLINE
   *    field       = key ":" value NEWLINE
   *    key         = [a-zA-Z_]+
   *    value       = unquoted-text | quoted-str
   *    quoted-str  = '"' [^"]* '"'
   *
   *  Fields:
   *    cover     – path or URL to cover image
   *    author    – author name (optional)
   *    start     – ISO date  YYYY-MM-DD
   *    status    – finished | ongoing | dropped | paused
   *    end       – ISO date  (required when status = finished)
   *    progress  – integer 0–100
   *    notes     – free-form text
   * ───────────────────────────────────────────── */

  class BSMLParser {
    constructor(src) {
      this.lines = src.split('\n');
      this.pos   = 0;
    }

    /* strip inline comment and trailing whitespace */
    clean(line) {
      // only strip # that is not inside a quoted value
      return line.replace(/\s*#[^"]*$/, '').trimEnd();
    }

    skipEmpty() {
      while (this.pos < this.lines.length) {
        const t = this.lines[this.pos].trim();
        if (t === '' || t.startsWith('#')) { this.pos++; }
        else break;
      }
    }

    unquote(s) {
      s = s.trim().replace(/[,;]\s*$/, '');   /* tolerate trailing , or ; */
      if (s.length >= 2 && s[0] === '"' && s[s.length - 1] === '"') {
        return s.slice(1, -1);
      }
      return s;
    }

    parse() {
      const shelf = { title: '', books: [] };
      while (this.pos < this.lines.length) {
        this.skipEmpty();
        if (this.pos >= this.lines.length) break;

        const line = this.clean(this.lines[this.pos]).trim();

        if (line.toLowerCase().startsWith('shelf ')) {
          shelf.title = this.unquote(line.slice(6).trim());
          this.pos++;
        } else if (line.toLowerCase().startsWith('book ')) {
          const book = this.parseBook(line);
          if (book) shelf.books.push(book);
        } else {
          this.pos++;
        }
      }
      return shelf;
    }

    parseBook(header) {
      /* header: book "Title" { */
      const m = header.match(/^book\s+"([^"]+)"\s*\{?\s*$/i);
      const book = {
        title:    m ? m[1] : 'Untitled',
        cover:    '',
        author:   '',
        start:    '',
        status:   'ongoing',
        end:      '',
        progress: 0,
        notes:    '',
      };
      this.pos++;

      while (this.pos < this.lines.length) {
        const raw   = this.lines[this.pos];
        const line  = this.clean(raw).trim();

        if (line === '}') { this.pos++; break; }
        if (line === '')  { this.pos++; continue; }

        const ci = line.indexOf(':');
        if (ci > 0) {
          const key = line.slice(0, ci).trim().toLowerCase();
          const val = line.slice(ci + 1).trim();
          if (key in book) book[key] = this.coerce(key, this.unquote(val));
        }
        this.pos++;
      }
      return book;
    }

    coerce(key, val) {
      if (key === 'progress') return Math.min(100, Math.max(0, parseInt(val) || 0));
      return val;
    }
  }

  /* ─── Date helpers ─── */

  function daysBetween(a, b) {
    return Math.round((b - a) / 86400000);
  }

  function duration(book) {
    if (!book.start) return null;
    const start = new Date(book.start);
    if (isNaN(start)) return null;
    const end = (book.status === 'finished' && book.end)
      ? new Date(book.end)
      : new Date();
    if (isNaN(end)) return null;

    const days = daysBetween(start, end);
    if (days < 0) return null;
    if (days === 0) return 'today';
    if (days < 7)  return days + 'd';
    if (days < 30) {
      const w = Math.floor(days / 7), d = days % 7;
      return d ? `${w}w ${d}d` : `${w}w`;
    }
    /* proper month diff */
    const y1 = start.getFullYear(), mo1 = start.getMonth(), d1 = start.getDate();
    const y2 = end.getFullYear(),   mo2 = end.getMonth(),   d2 = end.getDate();
    let months = (y2 - y1) * 12 + (mo2 - mo1);
    let rem    = d2 - d1;
    if (rem < 0) { months--; rem += 30; }
    if (months <= 0) return days + 'd';
    return rem > 0 ? `${months}mo ${rem}d` : `${months}mo`;
  }

  function fmtDate(d) {
    if (!d) return '';
    const dt = new Date(d);
    if (isNaN(dt)) return d;
    return dt.toLocaleDateString('en', { year: 'numeric', month: 'short', day: 'numeric' });
  }

  /* ─── HTML builder ─── */

  function esc(s) {
    return String(s)
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }

  function buildBook(book) {
    const dur    = duration(book);
    const status = book.status || 'ongoing';
    const statusCls = 'bsml-status-' + esc(status);

    /* progress bar */
    const pct = book.progress;

    /* tooltip rows */
    const rows = [];
    if (book.author) rows.push(['Author',   esc(book.author)]);
    if (book.start)  rows.push(['Started',  fmtDate(book.start)]);
    rows.push(['Status', `<span class="bsml-tip-badge ${statusCls}">${esc(status)}</span>`]);
    if (book.status === 'finished' && book.end) rows.push(['Finished', fmtDate(book.end)]);
    if (dur)         rows.push(['Reading',  esc(dur)]);
    rows.push(['Progress', `${pct}%`]);

    const rowsHTML = rows.map(([l, v]) =>
      `<div class="bsml-tip-row">
         <span class="bsml-tip-label">${l}</span>
         <span class="bsml-tip-value">${v}</span>
       </div>`
    ).join('');

    const notesHTML = book.notes
      ? `<div class="bsml-tip-notes">${esc(book.notes)}</div>` : '';

    return `
<div class="bsml-book ${statusCls}" tabindex="0" role="button"
     aria-label="${esc(book.title)}" aria-haspopup="true">
  <div class="bsml-cover-wrap">
    <img class="bsml-cover-img"
         src="${esc(book.cover)}"
         alt="${esc(book.title)}"
         loading="lazy"
         onerror="this.style.display='none';this.nextElementSibling.style.display='flex'">
    <div class="bsml-cover-fallback">${esc(book.title)}</div>
    <div class="bsml-progress-track">
      <div class="bsml-progress-fill" style="width:${pct}%"></div>
    </div>
  </div>
  ${dur ? `<div class="bsml-duration-badge">${esc(dur)}</div>` : ''}

  <div class="bsml-tooltip" role="tooltip">
    <div class="bsml-tip-title">${esc(book.title)}</div>
    ${rowsHTML}
    <div class="bsml-tip-progress-wrap">
      <div class="bsml-tip-progress-fill" style="width:${pct}%"></div>
    </div>
    ${notesHTML}
  </div>
</div>`;
  }

  function render(rootId) {
    const root    = document.getElementById(rootId);
    const srcEl   = document.getElementById(rootId + '-data');
    if (!root || !srcEl) return;

    const parser = new BSMLParser(srcEl.textContent);
    const shelf  = parser.parse();

    const titleHTML = shelf.title
      ? `<div class="bsml-shelf-title">${esc(shelf.title)}</div>` : '';

    const booksHTML = shelf.books.map(buildBook).join('\n');

    root.innerHTML = `
      ${titleHTML}
      <div class="bsml-shelf">
        <div class="bsml-books-row">${booksHTML}</div>
        <div class="bsml-plank"></div>
      </div>`;

    /* ── viewport edge-clamping ────────────────────────────────────────
     *  Measures the tooltip's default centred position, then shifts
     *  `left` by exactly the overflow amount so it stays on screen.
     *  The CSS variable --bsml-tip-arrow-dx is set to the same value
     *  so the arrow stays pointing at the book cover regardless.
     * ──────────────────────────────────────────────────────────────── */
    function clampTooltip(bookEl) {
      const tip = bookEl.querySelector('.bsml-tooltip');
      if (!tip) return;

      /* reset any previous run so we measure from the default position */
      tip.style.removeProperty('left');
      tip.style.removeProperty('--bsml-tip-arrow-dx');

      /* opacity:0 elements are still fully laid out — BoundingClientRect works */
      const tipRect = tip.getBoundingClientRect();
      const vw      = window.innerWidth || document.documentElement.clientWidth;
      const PAD     = 12;   /* minimum gap from either viewport edge */

      let overflow = 0;
      if      (tipRect.left  < PAD)       overflow = tipRect.left  - PAD;        /* < 0 */
      else if (tipRect.right > vw - PAD)  overflow = tipRect.right - (vw - PAD); /* > 0 */

      if (overflow !== 0) {
        /* shift the card: left overflow → increase left, right overflow → decrease left */
        tip.style.left = `calc(50% - ${overflow}px)`;
        /* compensate arrow so it still points at the book centre */
        tip.style.setProperty('--bsml-tip-arrow-dx', overflow + 'px');
      }
    }

    function resetTooltip(bookEl) {
      const tip = bookEl.querySelector('.bsml-tooltip');
      if (!tip) return;
      tip.style.removeProperty('left');
      tip.style.removeProperty('--bsml-tip-arrow-dx');
    }

    /* ── interaction: hover / focus / touch / keyboard ── */
    root.querySelectorAll('.bsml-book').forEach(el => {
      /* recalculate position every time the card is about to appear */
      el.addEventListener('mouseenter', () => clampTooltip(el));
      el.addEventListener('focus',      () => clampTooltip(el));
      el.addEventListener('mouseleave', () => resetTooltip(el));
      el.addEventListener('blur',       () => resetTooltip(el));

      /* touch / click toggle */
      el.addEventListener('click', e => {
        const alreadyOpen = el.classList.contains('bsml-tip-open');
        /* close all others and reset their positions */
        root.querySelectorAll('.bsml-tip-open').forEach(x => {
          x.classList.remove('bsml-tip-open');
          resetTooltip(x);
        });
        if (!alreadyOpen) {
          clampTooltip(el);
          el.classList.add('bsml-tip-open');
        }
        e.stopPropagation();
      });

      /* keyboard */
      el.addEventListener('keydown', e => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          el.click();
        }
        if (e.key === 'Escape') {
          el.classList.remove('bsml-tip-open');
          resetTooltip(el);
        }
      });
    });

    /* close tooltip when clicking outside */
    document.addEventListener('click', () => {
      root.querySelectorAll('.bsml-tip-open').forEach(x => {
        x.classList.remove('bsml-tip-open');
        resetTooltip(x);
      });
    });
  }

  /* run after DOM is ready */
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', () => render('{{ $id }}'));
  } else {
    render('{{ $id }}');
  }
})();
</script>
